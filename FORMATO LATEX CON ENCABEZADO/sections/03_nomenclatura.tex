\subsection*{Símbolos}

\begin{tabularx}{\textwidth}{@{}l>{\raggedright\arraybackslash}X@{}}
	\toprule
	\textbf{Símbolo} & \textbf{Descripción} \\
	\midrule
	$P$ & Presión estática manométrica (\SI{}{\kilo\pascal} / bar) \\
	$\Delta P$ & Pérdida de presión por fricción y elevación (\SI{}{\kilo\pascal} / bar) \\
	$Q$ & Caudal volumétrico (\SI{}{\cubic\meter\per\hour} / \SI{}{\liter\per\second}) \\
	$v$ & Velocidad media del fluido en la tubería (\SI{}{\meter\per\second}) \\
	$Re$ & Número de Reynolds (adimensional) \\
	$f$ & Factor de fricción de Darcy-Weisbach (adimensional) \\
	$h_L$ & Pérdida de energía hidráulica total expresada en cabeza de fluido (\SI{}{\meter}) \\
	$h_f$ & Pérdida de cabeza hidráulica por fricción primaria (\SI{}{\meter}) \\
	$h_m$ & Pérdida de cabeza hidráulica menor o secundaria por accesorios/curvas (\SI{}{\meter}) \\
	$K$ & Coeficiente de pérdidas menores (adimensional) \\
	$ID$ & Diámetro interno de la tubería (\SI{}{\meter} / \SI{}{\mm}) \\
	$OD$ & Diámetro exterior de la tubería (\SI{}{\meter} / \SI{}{\mm}) \\
	$\rho$ & Densidad del fluido (\SI{}{\kilogram\per\cubic\meter}) \\
	$\mu$ & Viscosidad dinámica del fluido (\SI{}{\milli\pascal\second}) \\
	$\epsilon$ & Rugosidad absoluta de la tubería (\SI{}{\meter} / \SI{}{\mm}) \\
	$dH$ & Caída o diferencia vertical de elevación (\SI{}{\meter}) \\
	\bottomrule
\end{tabularx}
